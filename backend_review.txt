import requests

BASE_URL = "http://127.0.0.1:5000"

# Step 1: Get Admin credentials
email = input("Enter Admin email: ").strip()
password = getpass.getpass("Enter Admin password: ")

try:
    # Step 2: Login
    login_response = requests.post(
        f"{BASE_URL}/api/auth/login",
        json={
            "email": email,
            "password": password
        },
        timeout=15
    )

    print("\nLogin status:", login_response.status_code)

    if login_response.status_code != 200:
        print("Login failed:", login_response.text)
        raise SystemExit

    # Step 3: Extract JWT token
    login_data = login_response.json()
    token = login_data.get("access_token")

    if not token:
        print("Login succeeded, but access_token was not returned.")
        print("Response keys:", list(login_data.keys()))
        raise SystemExit

    print("Admin authentication successful!")
    print("JWT token received.")

    # Step 4: Access protected Admin dashboard
    headers = {
        "Authorization": f"Bearer {token}"
    }

    dashboard_response = requests.get(
        f"{BASE_URL}/api/admin/dashboard",
        headers=headers,
        timeout=15
    )

    print("\nAdmin dashboard status:", dashboard_response.status_code)

    if dashboard_response.status_code == 200:
        print("Admin dashboard access successful!")
        print("Dashboard information:")
        print(dashboard_response.json())

    else:
        print("Dashboard access failed.")
        print(dashboard_response.text)

except requests.exceptions.RequestException as error:
    print("Connection error:", error)

from functools import wraps

from flask import jsonify, g
from flask_jwt_extended import (
    verify_jwt_in_request,
    get_jwt_identity
)

from db import get_db_connection


def role_required(*allowed_roles):

    def decorator(function):

        @wraps(function)
        def wrapper(*args, **kwargs):

            connection = None
            cursor = None

            try:
                # Step 1: Verify JWT token
                verify_jwt_in_request()

                # Step 2: Get logged-in user ID
                user_id = get_jwt_identity()

                # Step 3: Connect to database
                connection = get_db_connection()
                cursor = connection.cursor()

                # Step 4: Get current role and account status
                cursor.execute(
                    """
                    SELECT
                        users.user_id,
                        users.name,
                        users.email,
                        users.status,
                        roles.role_name
                    FROM users
                    JOIN roles
                    ON users.role_id = roles.role_id
                    WHERE users.user_id = %s
                    """,
                    (user_id,)
                )

                user = cursor.fetchone()

                # Step 5: Check user existence
                if not user:
                    return jsonify({
                        "message": "User account not found"
                    }), 401

                # Step 6: Check account status
                if user["status"] != "ACTIVE":
                    return jsonify({
                        "message": "Account is not active"
                    }), 403

                # Step 7: Check user role
                if user["role_name"] not in allowed_roles:
                    return jsonify({
                        "message": "You are not authorized to access this resource"
                    }), 403

                # Step 8: Store user details for the route
                g.current_user = user

            except Exception as error:
                from flask_jwt_extended.exceptions import JWTExtendedException

                if isinstance(error, JWTExtendedException):
                    return jsonify({
                        "message": "Invalid or expired authentication token"
                    }), 401

                print("Authorization error:", error)

                return jsonify({
                    "message": "Authorization service unavailable"
                }), 500

            finally:
                if cursor:
                    cursor.close()

                if connection and not connection.closed:
                    connection.close()

            # Step 9: Allow authorized request
            return function(*args, **kwargs)

        return wrapper

    return decorator

@admin_bp.route("/users/<int:user_id>/role", methods=["PATCH"])
@role_required("ADMIN")
def update_user_role(user_id):

    data = request.get_json(silent=True) or {}

    new_role = data.get("role", "")

    if not isinstance(new_role, str):
        return jsonify({
            "message": "Invalid role format"
        }), 400

    new_role = new_role.strip().upper()

    # Prevent creation of Admin accounts
    if new_role not in ["OWNER", "VIEWER"]:
        return jsonify({
            "message": "Only OWNER and VIEWER roles can be assigned here"
        }), 400

    # Prevent Admin from changing their own role
    if user_id == g.current_user["user_id"]:
        return jsonify({
            "message": "You cannot change your own role"
        }), 403

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Check target user
        cursor.execute(
            """
            SELECT user_id
            FROM users
            WHERE user_id = %s
            """,
            (user_id,)
        )

        target_user = cursor.fetchone()

        if not target_user:
            connection.rollback()

            return jsonify({
                "message": "User not found"
            }), 404

        # Get role ID
        cursor.execute(
            """
            SELECT role_id
            FROM roles
            WHERE role_name = %s
            """,
            (new_role,)
        )

        role = cursor.fetchone()

        if not role:
            connection.rollback()

            return jsonify({
                "message": "Role not found"
            }), 404

        # Update role
        cursor.execute(
            """
            UPDATE users
            SET role_id = %s
            WHERE user_id = %s
            """,
            (role["role_id"], user_id)
        )

        connection.commit()

        return jsonify({
            "message": "User role updated successfully",
            "user_id": user_id,
            "new_role": new_role
        }), 200

    except Exception as error:
        if connection and not connection.closed:
            connection.rollback()

        print("Update user role error:", error)

        return jsonify({
            "message": "Unable to update user role"
        }), 500

    finally:
        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

@admin_bp.route("/users/<int:user_id>/status", methods=["PATCH"])
@role_required("ADMIN")
def update_user_status(user_id):

    data = request.get_json(silent=True) or {}

    new_status = data.get("status", "")

    if not isinstance(new_status, str):
        return jsonify({
            "message": "Invalid account status format"
        }), 400

    new_status = new_status.strip().upper()

    allowed_statuses = [
        "ACTIVE",
        "INACTIVE",
        "SUSPENDED"
    ]

    if new_status not in allowed_statuses:
        return jsonify({
            "message": "Invalid account status"
        }), 400

    if user_id == g.current_user["user_id"]:
        return jsonify({
            "message": "You cannot change your own account status"
        }), 403

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Lock the target user row during this transaction
        cursor.execute(
            """
            SELECT
                users.user_id,
                users.status,
                roles.role_name
            FROM users
            JOIN roles
            ON users.role_id = roles.role_id
            WHERE users.user_id = %s
            FOR UPDATE
            """,
            (user_id,)
        )

        target_user = cursor.fetchone()

        if not target_user:
            connection.rollback()

            return jsonify({
                "message": "User not found"
            }), 404

        # Protect the last active Admin
        if (
            target_user["role_name"] == "ADMIN"
            and new_status != "ACTIVE"
            and target_user["status"] == "ACTIVE"
        ):

            cursor.execute(
                """
                SELECT COUNT(*) AS active_admins
                FROM users
                JOIN roles
                ON users.role_id = roles.role_id
                WHERE roles.role_name = 'ADMIN'
                  AND users.status = 'ACTIVE'
                """
            )

            count = cursor.fetchone()["active_admins"]

            if count <= 1:
                connection.rollback()

                return jsonify({
                    "message": "Cannot deactivate the last active Admin"
                }), 403

        # Update status
        cursor.execute(
            """
            UPDATE users
            SET status = %s
            WHERE user_id = %s
            """,
            (new_status, user_id)
        )

        connection.commit()

        return jsonify({
            "message": "Account status updated successfully",
            "user_id": user_id,
            "status": new_status
        }), 200

    except Exception as error:
        if connection and not connection.closed:
            connection.rollback()

        print("Update user status error:", error)

        return jsonify({
            "message": "Unable to update account status"
        }), 500

    finally:
        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

def is_valid_email(email):
    pattern = r"^[^@\s]+@[^@\s]+\.[^@\s]+$"
    return re.match(pattern, email) is not None


# --------------------------------------------------
# USER REGISTRATION
# --------------------------------------------------

@auth_bp.route("/register", methods=["POST"])
def register():

    data = request.get_json(silent=True) or {}

    name = data.get("name", "")
    email = data.get("email", "")
    password = data.get("password", "")
    role_name = data.get("role", "VIEWER")

    # Validate input types
    if not all(
        isinstance(value, str)
        for value in [name, email, password, role_name]
    ):
        return jsonify({
            "message": "Invalid input format."
        }), 400

    # Clean input
    name = name.strip()
    email = email.strip().lower()
    role_name = role_name.strip().upper()

    # Required fields
    if not name or not email or not password:
        return jsonify({
            "message": "All fields are required."
        }), 400

    # Validate email
    if not is_valid_email(email):
        return jsonify({
            "message": "Enter a valid email address."
        }), 400

    # Public registration only allows OWNER and VIEWER
    if role_name not in ["OWNER", "VIEWER"]:
        return jsonify({
            "message": "Invalid registration role. Only OWNER and VIEWER can register."
        }), 400

    # Validate password length
    if len(password) < 8:
        return jsonify({
            "message": "Password must contain at least 8 characters."
        }), 400

    # bcrypt supports passwords up to 72 UTF-8 bytes
    if len(password.encode("utf-8")) > 72:
        return jsonify({
            "message": "Password must not exceed 72 UTF-8 bytes."
        }), 400

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Check whether email already exists
        cursor.execute(
            """
            SELECT user_id
            FROM users
            WHERE email = %s
            """,
            (email,)
        )

        existing_user = cursor.fetchone()

        if existing_user:
            return jsonify({
                "message": "Email already registered."
            }), 409

        # Get role ID
        cursor.execute(
            """
            SELECT role_id
            FROM roles
            WHERE role_name = %s
            """,
            (role_name,)
        )

        role = cursor.fetchone()

        if not role:
            return jsonify({
                "message": "Selected role does not exist."
            }), 400

        # Hash password
        password_hash = bcrypt.hashpw(
            password.encode("utf-8"),
            bcrypt.gensalt()
        ).decode("utf-8")

        # Insert new user
        cursor.execute(
            """
            INSERT INTO users
            (
                name,
                email,
                password_hash,
                role_id
            )
            VALUES (%s, %s, %s, %s)
            """,
            (
                name,
                email,
                password_hash,
                role["role_id"]
            )
        )

        connection.commit()

        return jsonify({
            "success": True,
            "message": "Registration successful.",
            "role": role_name
        }), 201

    except Exception as error:

        if connection and not connection.closed:
            connection.rollback()

        print("Registration error:", error)
        traceback.print_exc()

        return jsonify({
            "success": False,
            "message": "Registration failed due to a server error."
        }), 500

    finally:

        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

@auth_bp.route("/login", methods=["POST"])
def login():

    data = request.get_json(silent=True) or {}

    email = data.get("email", "")
    password = data.get("password", "")

    # Validate input types
    if not isinstance(email, str) or not isinstance(password, str):
        return jsonify({
            "message": "Invalid input format."
        }), 400

    email = email.strip().lower()

    if not email or not password:
        return jsonify({
            "message": "Email and password are required."
        }), 400

    if len(password.encode("utf-8")) > 72:
        return jsonify({
            "message": "Invalid email or password."
        }), 401

    ip_address = request.remote_addr

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Find user and role
        cursor.execute(
            """
            SELECT
                users.user_id,
                users.name,
                users.email,
                users.password_hash,
                users.status,
                roles.role_name
            FROM users
            JOIN roles
                ON users.role_id = roles.role_id
            WHERE users.email = %s
            """,
            (email,)
        )

        user = cursor.fetchone()

        # Verify password safely
        password_valid = False

        if user:
            try:
                password_valid = bcrypt.checkpw(
                    password.encode("utf-8"),
                    user["password_hash"].encode("utf-8")
                )

            except (ValueError, TypeError, AttributeError):
                password_valid = False

        # Invalid email or password
        if not user or not password_valid:

            cursor.execute(
                """
                INSERT INTO login_attempts
                (
                    user_id,
                    email,
                    status,
                    ip_address
                )
                VALUES (%s, %s, 'FAILURE', %s)
                """,
                (
                    user["user_id"] if user else None,
                    email,

@owner_bp.route(
    "/documents/<int:document_id>/share",
    methods=["POST"]
)
@role_required("OWNER")
def share_document(document_id):

    data = request.get_json(silent=True) or {}

    viewer_email = data.get("viewer_email", "")

    if not isinstance(viewer_email, str):
        return jsonify({
            "message": "Invalid Viewer email"
        }), 400

    viewer_email = viewer_email.strip().lower()

    can_view = data.get("can_view", True)
    can_download = data.get("can_download", False)
    expires_at = data.get("expires_at")

    if not viewer_email:
        return jsonify({
            "message": "Viewer email is required"
        }), 400

    if not isinstance(can_view, bool) or not isinstance(can_download, bool):
        return jsonify({
            "message": "Permissions must be true or false"
        }), 400

    if not can_view:
        return jsonify({
            "message": "A shared document must allow viewing"
        }), 400

    # Validate optional expiry date
    if expires_at is not None:
        try:
            expiry = datetime.fromisoformat(
                expires_at.replace("Z", "+00:00")
            )

            if expiry.tzinfo is None:
                expiry = expiry.replace(tzinfo=timezone.utc)

            if expiry <= datetime.now(timezone.utc):
                return jsonify({
                    "message": "Expiry date must be in the future"
                }), 400

            expires_at = expiry.astimezone(
                timezone.utc
            ).replace(tzinfo=None)

        except (ValueError, AttributeError):
            return jsonify({
                "message": "Invalid expiry date"
            }), 400

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Verify document ownership
        cursor.execute(
            """
            SELECT document_id
            FROM documents
            WHERE document_id = %s
              AND owner_id = %s
              AND is_deleted = FALSE
            """,
            (
                document_id,
                g.current_user["user_id"]
            )
        )

        document = cursor.fetchone()

        if not document:
            connection.rollback()

            return jsonify({
                "message": "Document not found"
            }), 404

        # Find the active Viewer
        cursor.execute(
            """
            SELECT users.user_id
            FROM users
            JOIN roles
            ON users.role_id = roles.role_id
            WHERE users.email = %s
              AND roles.role_name = 'VIEWER'
              AND users.status = 'ACTIVE'
            """,
            (viewer_email,)
        )

        viewer = cursor.fetchone()

        if not viewer:
            connection.rollback()

            return jsonify({
                "message": "Active Viewer account not found"
            }), 404

        # Check for an existing active permission
        cursor.execute(
            """
            SELECT permission_id
            FROM document_permissions
            WHERE document_id = %s
              AND viewer_id = %s
              AND revoked_at IS NULL
            FOR UPDATE
            """,
            (document_id, viewer["user_id"])
        )

        existing_permission = cursor.fetchone()

        if existing_permission:
            connection.rollback()

            return jsonify({
                "message": "This document is already shared with this Viewer"
            }), 409

        # Create permission
        cursor.execute(
            """
            INSERT INTO document_permissions
            (
                document_id,
                viewer_id,
                can_view,
                can_download,
                expires_at
            )
            VALUES (%s, %s, %s, %s, %s)
            RETURNING permission_id
            """,
            (
                document_id,
                viewer["user_id"],
                can_view,
                can_download,
                expires_at
            )
        )

        permission_id = cursor.fetchone()["permission_id"]

        connection.commit()

        return jsonify({
            "message": "Document shared successfully",
            "permission_id": permission_id,
            "viewer_email": viewer_email,
            "can_view": can_view,
            "can_download": can_download
        }), 201

    except Exception as error:
        if connection and not connection.closed:
            connection.rollback()

        print("Share document error:", error)

        return jsonify({
            "message": "Unable to share document"
        }), 500

    finally:
        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

@viewer_bp.route(
    "/documents/<int:document_id>/view",
    methods=["GET"]
)
@role_required("VIEWER")
def view_shared_document(document_id):

    connection = None
    cursor = None

    viewer_id = g.current_user["user_id"]
    ip_address = request.remote_addr

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Step 1: Find the document
        cursor.execute(
            """
            SELECT
                document_id,
                document_name,
                storage_path
            FROM documents
            WHERE document_id = %s
              AND is_deleted = FALSE
            """,
            (document_id,)
        )

        document = cursor.fetchone()

        # Document does not exist or is archived
        if not document:

            record_audit_event(
                viewer_id,
                None,
                "DOCUMENT_VIEW",
                "DENIED",
                "DOCUMENT_NOT_AVAILABLE",
                ip_address
            )

            return jsonify({
                "success": False,
                "message": "Document is not available"
            }), 404

        # Step 2: Check active viewing permission
        cursor.execute(
            """
            SELECT permission_id
            FROM document_permissions
            WHERE document_id = %s
              AND viewer_id = %s
              AND can_view = TRUE
              AND revoked_at IS NULL
              AND (
                    expires_at IS NULL
                    OR expires_at > CURRENT_TIMESTAMP
              )
            LIMIT 1
            """,
            (document_id, viewer_id)
        )

        permission = cursor.fetchone()

        # Permission missing, revoked or expired
        if not permission:

            record_audit_event(
                viewer_id,
                document_id,
                "DOCUMENT_VIEW",
                "DENIED",
                "VIEW_PERMISSION_MISSING_OR_EXPIRED",
                ip_address
            )

            return jsonify({
                "success": False,
                "message": "You do not have permission to view this document"
            }), 403

        # Step 3: Retrieve the document from private Supabase Storage
        file_content = download_file(document["storage_path"])

        # Step 5: Record successful authorization
        record_audit_event(
            viewer_id,
            document_id,
            "DOCUMENT_VIEW",
            "ALLOWED",
            "VIEW_PERMISSION_VERIFIED",
            ip_address
        )

        # Step 6: Send document for viewing
        return send_file(
            BytesIO(file_content),
            as_attachment=False,
            download_name=document["document_name"],
            conditional=True
        )

    except Exception as error:

        print("Secure document viewing error:", error)

        return jsonify({

@viewer_bp.route(
    "/documents/<int:document_id>/download",
    methods=["GET"]
)
@role_required("VIEWER")
def download_shared_document(document_id):

    connection = None
    cursor = None

    viewer_id = g.current_user["user_id"]
    ip_address = request.remote_addr

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Step 1: Check whether document exists
        cursor.execute(
            """
            SELECT
                document_id,
                document_name,
                storage_path
            FROM documents
            WHERE document_id = %s
              AND is_deleted = FALSE
            """,
            (document_id,)
        )

        document = cursor.fetchone()

        if not document:

            record_audit_event(
                viewer_id,
                None,
                "DOCUMENT_DOWNLOAD",
                "DENIED",
                "DOCUMENT_NOT_AVAILABLE",
                ip_address
            )

            return jsonify({
                "success": False,
                "message": "Document is not available"
            }), 404

        # Step 2: Check viewing and downloading permissions
        cursor.execute(
            """
            SELECT permission_id
            FROM document_permissions
            WHERE document_id = %s
              AND viewer_id = %s
              AND can_view = TRUE
              AND can_download = TRUE
              AND revoked_at IS NULL
              AND (
                    expires_at IS NULL
                    OR expires_at > CURRENT_TIMESTAMP
              )
            LIMIT 1
            """,
            (document_id, viewer_id)
        )

        permission = cursor.fetchone()

        # Step 3: Reject unauthorized download
        if not permission:

            record_audit_event(
                viewer_id,
                document_id,
                "DOCUMENT_DOWNLOAD",
                "DENIED",
                "DOWNLOAD_PERMISSION_MISSING_OR_EXPIRED",
                ip_address
            )

            return jsonify({
                "success": False,
                "message": "You do not have permission to download this document"
            }), 403

        # Step 4: Retrieve the document from private Supabase Storage
        file_content = download_file(document["storage_path"])

        # Step 6: Record successful download authorization
        record_audit_event(
            viewer_id,
            document_id,
            "DOCUMENT_DOWNLOAD",
            "ALLOWED",
            "DOWNLOAD_PERMISSION_VERIFIED",
            ip_address
        )

        # Step 7: Send file as download
        return send_file(
            BytesIO(file_content),
            as_attachment=True,
            download_name=document["document_name"],
            conditional=True
        )

    except Exception as error:

        print("Secure document download error:", error)

        return jsonify({
            "success": False,
            "message": "Unable to download document"
        }), 500

    finally:

        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

import hashlib
import json
from datetime import datetime, timezone

from db import get_db_connection


def record_audit_event(
    user_id,
    document_id,
    action,
    result,
    reason,
    ip_address
):
    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # PostgreSQL advisory lock:
        # Ensures only one request writes to the audit chain at a time.
        cursor.execute(
            "SELECT pg_advisory_xact_lock(724681923)"
        )

        # Find the most recent audit hash
        cursor.execute("""
            SELECT current_hash
            FROM audit_logs
            ORDER BY log_id DESC
            LIMIT 1
        """)

        previous_record = cursor.fetchone()

        previous_hash = (
            previous_record["current_hash"]
            if previous_record
            else "0" * 64
        )

        # Use one timestamp for both the stored event and its hash
        event_time = datetime.now(timezone.utc).replace(tzinfo=None)

        event_data = {
            "user_id": user_id,
            "document_id": document_id,
            "action": action,
            "result": result,
            "reason": reason,
            "ip_address": ip_address,
            "timestamp": event_time.isoformat(timespec="microseconds"),
            "previous_hash": previous_hash
        }

        # Convert event details into consistent JSON
        event_json = json.dumps(
            event_data,
            sort_keys=True,
            separators=(",", ":")
        )

        # Generate SHA-256 hash
        current_hash = hashlib.sha256(
            event_json.encode("utf-8")
        ).hexdigest()

        # Insert the audit event
        cursor.execute("""
            INSERT INTO audit_logs (
                user_id,
                document_id,
                action,
                result,
                reason,
                ip_address,
                timestamp,
                previous_hash,
                current_hash
            )
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s)
        """, (
            user_id,
            document_id,
            action,
            result,
            reason,
            ip_address,
            event_time,
            previous_hash,
            current_hash
        ))

        # Commit also releases the transaction-level advisory lock
        connection.commit()

    except Exception:
        if connection:
            connection.rollback()
        raise

    finally:
        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

import hashlib
import json

from db import get_db_connection


def calculate_audit_hash(event_data):
    """
    Recalculate the SHA-256 hash using the same
    JSON format used when the audit record was created.
    """

    event_json = json.dumps(
        event_data,
        sort_keys=True,
        separators=(",", ":")
    )

    return hashlib.sha256(
        event_json.encode("utf-8")
    ).hexdigest()


def verify_audit_chain():

    connection = None
    cursor = None

    try:
        connection = get_db_connection()
        cursor = connection.cursor()

        # Retrieve audit records in chronological order
        cursor.execute("""
            SELECT
                log_id,
                user_id,
                document_id,
                action,
                result,
                reason,
                ip_address,
                timestamp,
                previous_hash,
                current_hash
            FROM audit_logs
            ORDER BY log_id ASC
        """)

        records = cursor.fetchall()

        expected_previous_hash = "0" * 64

        total_records = len(records)
        verified_records = 0

        for record in records:

            # Step 1: Verify previous hash connection
            if record["previous_hash"] != expected_previous_hash:

                return {
                    "valid": False,
                    "message": "Audit chain is broken",
                    "failed_log_id": record["log_id"],
                    "reason": "PREVIOUS_HASH_MISMATCH",
                    "total_records": total_records,
                    "verified_records": verified_records
                }

            # Step 2: Rebuild the original event data
            event_data = {
                "user_id": record["user_id"],
                "document_id": record["document_id"],
                "action": record["action"],
                "result": record["result"],
                "reason": record["reason"],
                "ip_address": record["ip_address"],
                "timestamp": record["timestamp"].isoformat(
                    timespec="microseconds"
                ),
                "previous_hash": record["previous_hash"]
            }

            # Step 3: Recalculate the hash
            recalculated_hash = calculate_audit_hash(event_data)

            # Step 4: Compare calculated and stored hashes
            if recalculated_hash != record["current_hash"]:

                return {
                    "valid": False,
                    "message": "Audit record has been modified",
                    "failed_log_id": record["log_id"],
                    "reason": "CURRENT_HASH_MISMATCH",
                    "total_records": total_records,
                    "verified_records": verified_records
                }

            # Move to the next record
            expected_previous_hash = record["current_hash"]

            verified_records += 1

        return {
            "valid": True,
            "message": "All audit records are valid",
            "total_records": total_records,
            "verified_records": verified_records
        }

    except Exception as error:

        print("Audit verification error:", error)
        raise

    finally:

        if cursor:
            cursor.close()

        if connection and not connection.closed:
            connection.close()

import os
import hashlib
import uuid
import mimetypes

from werkzeug.utils import secure_filename
from dotenv import load_dotenv
from supabase import create_client

load_dotenv()


# Allowed document extensions
ALLOWED_EXTENSIONS = {
    "pdf",
    "doc",
    "docx",
    "txt",
    "xlsx",
    "csv"
}


def get_supabase_client():
    """Create a Supabase client using backend environment variables."""

    supabase_url = os.getenv("SUPABASE_URL")
    supabase_key = os.getenv("SUPABASE_SECRET_KEY")

    if not supabase_url or not supabase_key:
        raise RuntimeError(
            "Supabase configuration is missing. "
            "Check SUPABASE_URL and SUPABASE_SECRET_KEY."
        )

    return create_client(supabase_url, supabase_key)


def get_storage_bucket():
    """Get the private Supabase Storage bucket name."""

    bucket = os.getenv("SUPABASE_BUCKET")

    if not bucket:
        raise RuntimeError("SUPABASE_BUCKET is not configured.")

    return bucket


def allowed_file(filename):
    """Check whether the file extension is allowed."""

    if not filename or "." not in filename:
        return False

    extension = filename.rsplit(".", 1)[1].lower()
    return extension in ALLOWED_EXTENSIONS


def generate_file_hash(file_content):
    """Generate a SHA-256 hash from file bytes."""

    return hashlib.sha256(file_content).hexdigest()


def validate_storage_path(storage_path):
    """Accept only simple filenames generated by this service."""

    if (
        not storage_path
        or not isinstance(storage_path, str)
        or storage_path in {".", ".."}
        or "/" in storage_path
        or "\\" in storage_path
    ):
        raise ValueError("Invalid document storage path.")


def save_uploaded_file(uploaded_file, upload_folder=None):
    """
    Upload a document to private Supabase Storage.

    upload_folder is retained for compatibility but is not used.
    """

    if not uploaded_file or not uploaded_file.filename:
        raise ValueError("No file was selected.")

    original_name = secure_filename(uploaded_file.filename)

    if not original_name or "." not in original_name:
        raise ValueError("Invalid filename.")

    if not allowed_file(original_name):
        raise ValueError("File type is not allowed.")

    extension = original_name.rsplit(".", 1)[1].lower()

    file_content = uploaded_file.read()

    if not file_content:
        raise ValueError("The selected file is empty.")

    unique_filename = f"{uuid.uuid4().hex}.{extension}"

    file_hash = generate_file_hash(file_content)

    content_type = (
        mimetypes.guess_type(original_name)[0]
        or "application/octet-stream"
    )

    supabase = get_supabase_client()
    bucket = get_storage_bucket()

    try:
        supabase.storage.from_(bucket).upload(
            path=unique_filename,
            file=file_content,
            file_options={
                "content-type": content_type,
                "upsert": "false"
            }
        )

    except Exception as error:
        raise RuntimeError(
            "Document upload failed. Please try again."
        ) from error

    return {
        "original_name": original_name,
        "storage_name": unique_filename,
        "storage_path": unique_filename,
        "file_path": unique_filename,
        "file_hash": file_hash
    }


def download_file(storage_path):
    """Download a document from Supabase Storage as bytes."""

    validate_storage_path(storage_path)

    supabase = get_supabase_client()
    bucket = get_storage_bucket()

    try:
        return supabase.storage.from_(bucket).download(storage_path)

    except Exception as error:
        raise RuntimeError(
            "Unable to download the document from storage."
        ) from error


def delete_file(storage_path):
    """Delete a document from Supabase Storage."""

    validate_storage_path(storage_path)

    supabase = get_supabase_client()
    bucket = get_storage_bucket()

    try:
        supabase.storage.from_(bucket).remove([storage_path])
        return True

    except Exception as error:
        raise RuntimeError(
            "Unable to delete the document from storage."
        ) from error


def create_document_signed_url(storage_path, expires_in=60):
    """Generate a temporary URL for a private document."""

    validate_storage_path(storage_path)

    if not isinstance(expires_in, int) or expires_in <= 0:
        raise ValueError("Invalid signed URL expiry time.")

    supabase = get_supabase_client()
    bucket = get_storage_bucket()

    try:
        result = supabase.storage.from_(bucket).create_signed_url(
            storage_path,
            expires_in
        )

        return result.get("signedURL") or result.get("signedUrl")

    except Exception as error:
        raise RuntimeError(
            "Unable to generate document access URL."
        ) from error
